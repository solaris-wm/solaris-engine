# Extra dependencies for postprocess/render_voxels.py (voxel renderer) and the
# --voxel-cube / --voxel-pov modes of annotate_video_batch.py. Kept separate
# from env.yaml so the base postprocess pipeline runs without pytorch3d, which
# render_voxels imports lazily.
#
# Install on a Linux GPU machine. torch must be pinned to a pytorch3d-supported
# version (the latest torch is too new); then install the matching prebuilt
# pytorch3d wheel (no source build).
#
# Example for CUDA 12.1 + Python 3.10 (adjust cu121/pyt241 to your CUDA version
# and the torch you install; check `nvidia-smi`):
#
#   pip install torch==2.4.1 --index-url https://download.pytorch.org/whl/cu121
#   pip install --no-index pytorch3d \
#     -f https://dl.fbaipublicfiles.com/pytorch3d/packaging/wheels/py310_cu121_pyt241/download.html
#
# Alternative that resolves a compatible stack automatically:
#   conda install pytorch3d -c pytorch3d
torch==2.4.1
# pytorch3d is installed from the wheel index above (no plain PyPI wheel exists
# for this torch/CUDA/python combination).
